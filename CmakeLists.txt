cmake_minimum_required(VERSION 3.16)


project(CaIEngine)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

option(CAIENGINE_TEST "Whether to start test" ON)
if (APPLE)
    # 对于 macOS，可能需要
    set(CMAKE_CXX_FLAGS_DEBUG "-g -O0 -fno-limit-debug-info")
endif ()


file(GLOB scenePath CONFIGURE_DEPENDS
        "${CMAKE_CURRENT_SOURCE_DIR}/scene/*.h"
        "${CMAKE_CURRENT_SOURCE_DIR}/scene/*.cpp"
)

add_subdirectory(FrameWork)

add_executable(${PROJECT_NAME} Renderer.cpp ${scenePath}
)

if (CAIENGINE_TEST)
    enable_testing()
    add_subdirectory(test)
endif ()

target_link_libraries(${PROJECT_NAME} FrameWork)

target_include_directories(${PROJECT_NAME} PUBLIC FrameWork)
