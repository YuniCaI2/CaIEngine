cmake_minimum_required(VERSION 3.16)

set(CMAKE_CXX_STANDARD 23)
project(FrameWork)

find_package(Vulkan REQUIRED)
find_package(assimp CONFIG REQUIRED)
find_package(glfw3 CONFIG REQUIRED)
find_package(EnTT CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(OpenSSL REQUIRED)

if(WIN32)
    find_package(directxtex CONFIG REQUIRED)
endif()

add_subdirectory(thirdPart)

file(GLOB frameWorkPath CONFIGURE_DEPENDS
        "${CMAKE_CURRENT_SOURCE_DIR}/*.h"
        "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp"
)

file(GLOB frameGraphPath CONFIGURE_DEPENDS
        "${CMAKE_CURRENT_SOURCE_DIR}/FrameGraph/*.h"
        "${CMAKE_CURRENT_SOURCE_DIR}/FrameGraph/*.cpp"
)

file(GLOB uniformPassPath CONFIGURE_DEPENDS
        "${CMAKE_CURRENT_SOURCE_DIR}/FrameGraph/UniformPass/*.h"
        "${CMAKE_CURRENT_SOURCE_DIR}/FrameGraph/UniformPass/*.cpp"
)

file(GLOB ECSPath CONFIGURE_DEPENDS
        "${CMAKE_CURRENT_SOURCE_DIR}/ECS/Components/*.h"
        "${CMAKE_CURRENT_SOURCE_DIR}/ECS/Components/*.cpp"
        "${CMAKE_CURRENT_SOURCE_DIR}/ECS/Systems/*.cpp"
        "${CMAKE_CURRENT_SOURCE_DIR}/ECS/Systems/*.h"
        "${CMAKE_CURRENT_SOURCE_DIR}/ECS/resources/*.cpp"
        "${CMAKE_CURRENT_SOURCE_DIR}/ECS/resources/*.h"
)

file(GLOB EditorPath CONFIGURE_DEPENDS
        "${CMAKE_CURRENT_SOURCE_DIR}/Editor/*.h"
        "${CMAKE_CURRENT_SOURCE_DIR}/Editor/*.cpp"

)

file(GLOB AssetPath
        "${CMAKE_CURRENT_SOURCE_DIR}/Asset/*.h"
)

add_library(FrameWork ${frameWorkPath} ${frameGraphPath} ${uniformPassPath} ${ECSPath} ${AssetPath} ${EditorPath}
        CaIMaterial.cpp
        CaIMaterial.h
        CaIShader.cpp
        CaIShader.h
        FrameGraph/ResourceManager.cpp
        FrameGraph/ResourceManager.h
        FrameGraph/RenderPassManager.cpp
        FrameGraph/RenderPassManager.h
        FrameGraph/FrameGraph.cpp
        FrameGraph/FrameGraph.h
        CompMaterial.cpp
        CompMaterial.h
        CompShader.cpp
        CompShader.h
        Editor/Panel.h
)
if (MSVC)
    target_compile_options(FrameWork PRIVATE /bigobj)
endif()

# 基础库链接（所有平台共用）
target_link_libraries(FrameWork PUBLIC
        glfw
        Vulkan::Vulkan
        assimp::assimp
        EnTT::EnTT
        nlohmann_json::nlohmann_json
        OpenSSL::SSL
        OpenSSL::Crypto
)

# 平台特定的库和配置
if(WIN32)
    target_link_libraries(FrameWork PRIVATE Microsoft::DirectXTex)
elseif(APPLE)
endif()
target_link_libraries(FrameWork PUBLIC imgui)

# 第三方库头文件路径
target_include_directories(FrameWork PUBLIC
        ${CMAKE_CURRENT_SOURCE_DIR}/thirdPart/glm
        ${CMAKE_CURRENT_SOURCE_DIR}/thirdPart/stb
        ${CMAKE_CURRENT_SOURCE_DIR}/thirdPart/imgui
)

# 保证target_link_library可以找到FrameWork
target_include_directories(FrameWork INTERFACE
        ${CMAKE_CURRENT_SOURCE_DIR}/
)
